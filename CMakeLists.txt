cmake_minimum_required(VERSION 3.20)
project(APUE LANGUAGES C)
if(NOT UNIX)
    message(FATAL_ERROR "These POSIX examples require Linux/macOS; use WSL on Windows")
endif()
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
option(ENABLE_SANITIZERS "Enable address and undefined behavior sanitizers" OFF)

add_library(apue_options INTERFACE)
target_include_directories(apue_options INTERFACE "${PROJECT_SOURCE_DIR}")
target_compile_definitions(apue_options INTERFACE _POSIX_C_SOURCE=200809L)
if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(apue_options INTERFACE -Wall -Wextra -Wpedantic)
    if(ENABLE_SANITIZERS)
        target_compile_options(apue_options INTERFACE -fsanitize=address,undefined -fno-omit-frame-pointer)
        target_link_options(apue_options INTERFACE -fsanitize=address,undefined)
    endif()
endif()

add_library(apue_error STATIC lib/error.c)
target_link_libraries(apue_error PUBLIC apue_options)
if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(apue_error PRIVATE -Werror)
endif()

# Keep the book's source names; build only the verified introductory examples.
foreach(example 1-1 1-2 1-3 1-4 1-6 1-7 3-1 3-2 3-3 3-4)
    add_executable(example-${example} ${example}.c)
    target_link_libraries(example-${example} PRIVATE apue_error)
    if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
        target_compile_options(example-${example} PRIVATE -Werror)
    endif()
endforeach()

include(CTest)
if(BUILD_TESTING)
    find_package(Python3 3.10 REQUIRED COMPONENTS Interpreter)
    add_executable(test-error tests/error_main.c tests/error_helper.c)
    target_link_libraries(test-error PRIVATE apue_error)
    if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
        target_compile_options(test-error PRIVATE -Werror)
    endif()
    foreach(case errors examples)
        add_test(NAME ${case} COMMAND "${Python3_EXECUTABLE}" "${PROJECT_SOURCE_DIR}/tests/regression.py"
            "${CMAKE_RUNTIME_OUTPUT_DIRECTORY}" ${case})
        set_tests_properties(${case} PROPERTIES TIMEOUT 60
            FAIL_REGULAR_EXPRESSION "AddressSanitizer;UndefinedBehaviorSanitizer;runtime error:;LeakSanitizer")
    endforeach()
endif()
